def init_db():
    conn = sqlite3.connect('library.db')
    c = conn.cursor()
    c.execute('''CREATE TABLE IF NOT EXISTS borrows
                 (id INTEGER PRIMARY KEY AUTOINCREMENT, 
                  user_id INTEGER, chat_id INTEGER, 
                  name TEXT, phone TEXT, book_title TEXT, 
                  borrow_date DATE, status TEXT, reminder_sent INTEGER)''')
    conn.commit()
    conn.close()

user_steps = {}
user_data = {}

def main_keyboard():
    markup = types.ReplyKeyboardMarkup(resize_keyboard=True)
    markup.add(types.KeyboardButton('📚 امانت'), types.KeyboardButton('🔄 بازگشت'))
    markup.add(types.KeyboardButton('⏳ تمدید'), types.KeyboardButton('💡 پیشنهادات'))
    markup.add(types.KeyboardButton('💳 مشارکت مالی در توسعه کتابخانه'))
    return markup

@bot.message_handler(commands=['start'])
def send_welcome(message):
    bot.send_message(message.chat.id, "سلام! به ربات مدیریت کتابخانه خوش آمدید. لطفا یک گزینه را انتخاب کنید:", reply_markup=main_keyboard())

@bot.message_handler(func=lambda message: True)
def handle_messages(message):
    chat_id = message.chat.id
    text = message.text

    if text == '📚 امانت':
        user_steps[chat_id] = 'get_name'
        bot.send_message(chat_id, "لطفاً نام و نام خانوادگی خود را وارد کنید:", reply_markup=types.ReplyKeyboardRemove())
        
    elif text == '🔄 بازگشت':
        conn = sqlite3.connect('library.db')
        c = conn.cursor()
        c.execute("SELECT id, book_title FROM borrows WHERE chat_id=? AND status='borrowed'", (chat_id,))
        books = c.fetchall()
        conn.close()
        if books:
            markup = types.InlineKeyboardMarkup()
            for book in books:
                markup.add(types.InlineKeyboardButton(book[1], callback_data=f"return_{book[0]}"))
            bot.send_message(chat_id, "کدام کتاب را می‌خواهید بازگردانید؟", reply_markup=markup)
        else:
            bot.send_message(chat_id, "شما در حال حاضر کتاب امانتی ندارید.", reply_markup=main_keyboard())

    elif text == '⏳ تمدید':
        conn = sqlite3.connect('library.db')
        c = conn.cursor()
        c.execute("SELECT id, book_title FROM borrows WHERE chat_id=? AND status='borrowed'", (chat_id,))
        books = c.fetchall()
        if books:
            markup = types.InlineKeyboardMarkup()
            for book in books:
                markup.add(types.InlineKeyboardButton(book[1], callback_data=f"extend_{book[0]}"))
            bot.send_message(chat_id, "کدام کتاب را می‌خواهید تمدید کنید؟ (۵ روز به مهلت اضافه می‌شود)", reply_markup=markup)
        else:
            bot.send_message(chat_id, "کتابی برای تمدید وجود ندارد.", reply_markup=main_keyboard())
        conn.close()

    elif text == '💡 پیشنهادات':
        user_steps[chat_id] = 'get_suggestion'
        bot.send_message(chat_id, "لطفاً پیشنهاد یا انتقاد خود را بنویسید:", reply_markup=types.ReplyKeyboardRemove())

    elif text == '💳 مشارکت مالی در توسعه کتابخانه':
        bot.send_message(chat_id, "از اینکه در مسیر توسعه کتابخانه همراه ما هستید سپاسگزاریم. 🌸\n\nشماره کارت:\n`6037990000000000`\nبه نام: توسعه کتابخانه", parse_mode='Markdown', reply_markup=main_keyboard())

    elif chat_id in user_steps:
        step = user_steps[chat_id]
        if step == 'get_name':
            user_data[chat_id] = {'name': text}
            user_steps[chat_id] = 'get_phone'
            bot.send_message(chat_id, "شماره تماس خود را وارد کنید:")
        elif step == 'get_phone':
            user_data[chat_id]['phone'] = text
            user_steps[chat_id] = 'get_book'
            bot.send_message(chat_id, "عنوان کتابی که امانت می‌گیرید را بنویسید:")
        elif step == 'get_book':
            book_title = text
            data = user_data[chat_id]
            borrow_date = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            conn = sqlite3.connect('library.db')
            c = conn.cursor()
            c.execute("INSERT INTO borrows (user_id, chat_id, name, phone, book_title, borrow_date, status, reminder_sent) VALUES (?, ?, ?, ?, ?, ?, ?, ?)",
                      (message.from_user.id, chat_id, data['name'], data['phone'], book_title, borrow_date, 'borrowed', 0))
            conn.commit()
            conn.close()
            del user_steps[chat_id]
            bot.send_message(chat_id, f"✅ کتاب «{book_title}» با موفقیت برای شما امانت ثبت شد. مهلت مطالعه: ۵ روز.", reply_markup=main_keyboard())
        elif step == 'get_suggestion':
            bot.send_message(chat_id, "ممنون از پیشنهاد شما! بررسی خواهد شد.", reply_markup=main_keyboard())
            del user_steps[chat_id]
        elif step == 'get_feedback':
            bot.send_message(chat_id, "از اینکه نظر ارزشمندتان را با ما به اشتراک گذاشتید متشکریم! 🌹", reply_markup=main_keyboard())
            del user_steps[chat_id]

@bot.callback_query_handler(func=lambda call: True)
def handle_query(call):
    chat_id = call.message.chat.id
    action, record_id = call.data.split('_')
    conn = sqlite3.connect('library.db')
    c = conn.cursor()
    if action == 'return':
        c.execute("UPDATE borrows SET status='returned' WHERE id=?", (record_id,))
        bot.edit_message_text(chat_id=chat_id, message_id=call.message.message_id, text="✅ بازگشت کتاب با موفقیت ثبت شد.")
        user_steps[chat_id] = 'get_feedback'
        bot.send_message(chat_id, "اگر دوست دارید نظرتان را در مورد این کتاب اعلام کنید تا بقیه هم استفاده کنند:")
    elif action == 'extend':
        new_date = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        c.execute("UPDATE borrows SET borrow_date=?, reminder_sent=0 WHERE id=?", (new_date, record_id))
        bot.edit_message_text(chat_id=chat_id, message_id=call.message.message_id, text="✅ مهلت کتاب شما با موفقیت ۵ روز دیگر تمدید شد.")
    conn.commit()
    conn.close()

def check_overdue_books():
    conn = sqlite3.connect('library.db')
    c = conn.cursor()
    c.execute("SELECT id, chat_id, borrow_date FROM borrows WHERE status='borrowed' AND reminder_sent=0")
    records = c.fetchall()
    now = datetime.now()
    for row in records:
        record_id, chat_id, borrow_date_str = row
        borrow_date = datetime.strptime(borrow_date_str, "%Y-%m-%d %H:%M:%S")
        if now - borrow_date >= timedelta(days=5):
            msg = "سلام ارادت مدت 5 روزه مطالعه کتاب شما به اتمام رسیده اگر دوستدارید کتاب رو تمدید کنید از همینجا اقدام کنید و اگر نه در میدان مفید منتظر شما هستیم"
            try:
                bot.send_message(chat_id, msg)
                c.execute("UPDATE borrows SET reminder_sent=1 WHERE id=?", (record_id,))
                conn.commit()
            except:
                pass
    conn.close()

def run_scheduler():
    schedule.every(1).hours.do(check_overdue_books)
    while True:
        schedule.run_pending()
        time.sleep(60)

def run_bot():
    bot.infinity_polling()

if __name__ == "__main__":
    threading.Thread(target=run_bot, daemon=True).start()
    threading.Thread(target=run_scheduler, daemon=True).start()
    # اجرای سرور وب برای رندر
    port = int(os.environ.get("PORT", 10000))
    app.run(host='0.0.0.0', port=port)
